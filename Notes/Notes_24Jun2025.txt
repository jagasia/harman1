package com.harman.demo.model;

import jakarta.persistence.Entity;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.Id;
import jakarta.persistence.Table;

@Entity
@Table(name="CUST")
public class Customer {
	@Id
	@GeneratedValue
	private Long id;
	private String name;
	private String address;
	private String phone;
	
	public Customer() {}

	public Customer(Long id, String name, String address, String phone) {
		super();
		this.id = id;
		this.name = name;
		this.address = address;
		this.phone = phone;
	}

	public Long getId() {
		return id;
	}

	public void setId(Long id) {
		this.id = id;
	}

	public String getName() {
		return name;
	}

	public void setName(String name) {
		this.name = name;
	}

	public String getAddress() {
		return address;
	}

	public void setAddress(String address) {
		this.address = address;
	}

	public String getPhone() {
		return phone;
	}

	public void setPhone(String phone) {
		this.phone = phone;
	}

	@Override
	public String toString() {
		return "Customer [id=" + id + ", name=" + name + ", address=" + address + ", phone=" + phone + "]";
	}
	
}

package com.harman.demo.model;

import java.util.List;

import org.hibernate.Session;
import org.hibernate.SessionFactory;
import org.hibernate.Transaction;
import org.hibernate.query.Query;

import com.harman.demo.util.DbUtil;

public class CustomerDAOImpl implements CustomerDAO {

	@Override
	public void create(Customer customer) {
		SessionFactory sf = DbUtil.getSessionFactory();
		Session session = sf.openSession();
		Transaction txn = session.beginTransaction();
//		session.persist(customer);
		Object id = session.save(customer);
		System.out.println("Customer created. Kindly note the ID: ");
		System.out.println(id);
		txn.commit();
		session.clear();
	}
	@Override
	public List<Customer> read() {
		SessionFactory sf = DbUtil.getSessionFactory();
		Session session = sf.openSession();
		Query<Customer> query = session.createQuery("select c from Customer c",Customer.class);
		return query.list();
	}
	
	@Override
	public Customer read(Long id) {
		SessionFactory sf = DbUtil.getSessionFactory();
		Session session = sf.openSession();
		Customer managed = session.get(Customer.class, id);
		return managed;
	}
	
	@Override
	public void update(Customer customer) {
		SessionFactory sf = DbUtil.getSessionFactory();
		Session session = sf.openSession();
		Transaction txn = session.beginTransaction();
//		session.persist(customer);	 //detached entity error
//		detached entity passed to persist: com.harman.demo.model.Customer
		session.merge(customer);
		txn.commit();
		session.close();
	}
	@Override
	public void delete(Long id) {
		SessionFactory sf = DbUtil.getSessionFactory();
		Session session = sf.openSession();
		Transaction txn = session.beginTransaction();
		Customer x = session.get(Customer.class, id);
		if(x!=null) {
			session.delete(x);
		}else {
			System.out.println("Nothing to delete");
		}
		
		txn.commit();
		session.close();
	}
	
}

## 🔍 Hibernate Query Language (HQL) Tutorial

### ✅ What is HQL?

**HQL (Hibernate Query Language)** is an object-oriented query language similar to SQL but operates on **entity objects** (Java classes) rather than directly on database tables.

* SQL works with tables and columns.
* HQL works with **entities and their properties**.

---

### 🧱 Basic Syntax

```java
String hql = "FROM Employee";
Query query = session.createQuery(hql);
List<Employee> employees = query.list();
```

---

### 🧭 Common HQL Queries

#### 1. **Select all records**

```java
String hql = "FROM Employee";
List<Employee> list = session.createQuery(hql).list();
```

#### 2. **Select with WHERE clause**

```java
String hql = "FROM Employee e WHERE e.salary > :minSalary";
Query query = session.createQuery(hql);
query.setParameter("minSalary", 50000);
List<Employee> list = query.list();
```

#### 3. **Select specific columns (projection)**

```java
String hql = "SELECT e.name, e.salary FROM Employee e";
List<Object[]> list = session.createQuery(hql).list();

for (Object[] row : list) {
    String name = (String) row[0];
    Double salary = (Double) row[1];
}
```

#### 4. **Order By**

```java
String hql = "FROM Employee e ORDER BY e.salary DESC";
List<Employee> list = session.createQuery(hql).list();
```

#### 5. **Update Query**

```java
String hql = "UPDATE Employee SET salary = :salary WHERE id = :id";
Query query = session.createQuery(hql);
query.setParameter("salary", 60000);
query.setParameter("id", 2);
int rowsAffected = query.executeUpdate();
```

#### 6. **Delete Query**

```java
String hql = "DELETE FROM Employee WHERE id = :id";
Query query = session.createQuery(hql);
query.setParameter("id", 3);
int rowsAffected = query.executeUpdate();
```

#### 7. **Aggregate Functions**

```java
String hql = "SELECT COUNT(e.id), AVG(e.salary) FROM Employee e";
Object[] result = (Object[]) session.createQuery(hql).uniqueResult();

Long count = (Long) result[0];
Double averageSalary = (Double) result[1];
```

---

### 🔄 Joins in HQL

#### 1. **Inner Join**

```java
String hql = "SELECT e.name, d.name FROM Employee e INNER JOIN e.department d";
List<Object[]> list = session.createQuery(hql).list();
```

#### 2. **Left Join**

```java
String hql = "SELECT e.name, d.name FROM Employee e LEFT JOIN e.department d";
```

> Note: This assumes `Employee` has a field `Department department` with `@ManyToOne`.

---

### 📦 Pagination

```java
Query query = session.createQuery("FROM Employee");
query.setFirstResult(0);  // offset
query.setMaxResults(10);  // limit
List<Employee> list = query.list();
```

---

### ⚠️ Tips

* Always use **entity names and property names** in HQL.
* For **native SQL**, use `session.createSQLQuery(...)`.
* HQL supports **named queries** and **parameterized queries**.

---

### 📘 Example Entity

```java
@Entity
public class Employee {
    @Id
    private int id;
    private String name;
    private double salary;

    @ManyToOne
    private Department department;
}
```

Basically, when we do projects, it is important to identify the entities and their relationships

E-R diagrams can be the artifacts, that evolves at the end of this activity on identifying the entities and relationships

Entity becomes table in database
Entity becomes class in java code.

If 2 entities have relationship, then it can be
	One to One
	One to Many
	Many to One
	Many to Many


These relationships can be exactly represented in our code using annotations like
	@OneToOne
	@OneToMany
	@ManyToOne
	@ManyToMany

Why these are important?
	when we create tables, we will also create constraints like foreign key, that establishes the relationship between 2 tables. Parent-child or Master-details or Master-transaction etc.

Now that, Hibernate is going to create the tables, we need to help hibernate to understand the relationship, like what is foreign key that references which primary key of which table etc.

Apart from this, lot of advantages in having these relationships, lazy or eager loading, cascading, etc.


 Introduction to Hibernate Relationships
In real-world applications, data in a database is often related. For example:

A user might have one profile.

A department has many employees.

A student can register for many courses, and a course can have many students.

These relationships between tables are known as associations or entity relationships in Hibernate (and JPA).

Hibernate lets you map Java classes (entities) to database tables and also define how those classes relate to each other — just like relationships in a relational database (MySQL, PostgreSQL, etc.).

📘 Why Relationships Are Important
They let you represent real-world data structures (e.g., one user has many orders).

You can navigate through related data easily using Java objects.

They help maintain data integrity with constraints like foreign keys.

Hibernate handles automated joins, cascading, and fetching — reducing manual SQL.

🔗 Types of Relationships in Hibernate
Hibernate supports four main types of relationships:

Relationship Type	Real-Life Example	Description
One-to-One	A person and their passport	Each entity has exactly one related entity.
One-to-Many	A department and its employees	One entity is related to many entities.
Many-to-One	Many employees belong to one department	Inverse of One-to-Many.
Many-to-Many	Students and courses	Many entities are related to many others.

Each of these can be:

Unidirectional: Navigation is allowed in only one direction.

Bidirectional: Navigation is allowed in both directions (more common).

		txn.commit();
		session.close();

	@Override

	@Override
	public Employee read(Long id) {
		Session session = DbUtil.getSessionFactory()
				.openSession();
		Employee employee = session.get(Employee.class, id);
		session.close();
		return employee;
	}
		
	@Override
	public void update(Employee employee) {
		Session session = DbUtil.getSessionFactory()
				.openSession();
				Transaction txn = session.beginTransaction();
				Object id = session.merge(employee);
				txn.commit();
				session.close();
				System.out.println("Employee is updated");		
	}
	@Override
	public void delete(Long id) {
		Session session = DbUtil.getSessionFactory()
				.openSession();
				Transaction txn = session.beginTransaction();
				Employee employee = session.get(Employee.class, id);
				if(employee!=null) {
				session.delete(employee);
				}else
				{
					System.out.println("Nothing to delete");
				}
				
				session.close();
	}
