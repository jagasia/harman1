package com.harman.demo.service;

import java.util.List;
import java.util.Optional;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import com.harman.demo.entity.Product;
import com.harman.demo.exception.ProductNotFoundException;
import com.harman.demo.repository.ProductRepository;

@Service
public class ProductService {
	@Autowired
	private ProductRepository pr;
	
	public Product create(Product product) {
		return pr.save(product);
	}
	
	public List<Product> read() {
		return pr.findAll();
	}
	
	public Product read(Long id) {
		Optional<Product> temp = pr.findById(id);
		Product product=null;
		if(temp.isPresent()) {
			product=temp.get();
		}else {
			throw new ProductNotFoundException(id);
		}
		return product;
	}
	
	public Product update(Long id, Product product) {
		Product temp=read(id);
		if(temp!=null) {
			pr.save(product);
		}
		return temp;
	}
	
	public Product delete(Long id) {
		Product temp=read(id);
		if(temp!=null) {
			pr.delete(temp);
		}
		return temp;
	}
	
	
}


6) create the controller:

package com.harman.demo.controller;

import java.util.List;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Controller;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestMethod;
import org.springframework.web.servlet.ModelAndView;

import com.harman.demo.entity.Product;
import com.harman.demo.service.ProductService;

@Controller
@RequestMapping("/product")
public class ProductController {
	@Autowired
	private ProductService ps;
	
	@RequestMapping(value = "/dml", method = RequestMethod.POST, params = "add")
	public ModelAndView addProduct(Product product) {
		ps.create(product);
		return showProductPage();
	}
	
	
	@GetMapping
	public ModelAndView showProductPage() {
		List<Product> products = ps.read();
		ModelAndView mv=new ModelAndView();
		mv.setViewName("product");
		mv.addObject("products",products);
		return mv;
	}
	
	@GetMapping("/select/{id}")
	public ModelAndView findProductById(@PathVariable("id")Long id) {
		Product product = ps.read(id);
		List<Product> products = ps.read();
		ModelAndView mv=new ModelAndView();
		mv.setViewName("product");
		mv.addObject("products",products);
		mv.addObject("product",product);
		return mv;
	}
	
	@RequestMapping(value = "/dml", method = RequestMethod.POST, params = "update")
	public ModelAndView updateProduct(Product product) {
		ps.update(product.getId(), product);
		return showProductPage();
	}
	
	@RequestMapping(value = "/dml", method = RequestMethod.POST, params = "delete")
	public ModelAndView deleteProduct(Product product) {
		ps.delete(product.getId());
		return showProductPage();
	}
	
	
}

@Controller vs @RestController

@Controller
	is used in MVC applications
	in a controller class, a method returns
		view	(or)
		ModelAndView

	but we can modify that behaviour using @ResponseBody

ex:
in a controller class, the following method:

	@GetMapping("/hello")
	@ResponseBody		//note this 
	public String hello(){
		return "Hello world";
	}


@ReponseBody annotation over a method, will change what is returned by the method.
	instead of returning the view name, the method returns the response body that is going to be printed in the browser.


In @RestController, the methods return response body only. We do not need to mention @ResponseBody on the methods. 


@RestController	=>	@Controller + @ResponseBody

	@GetMapping
	public String home() {
		return "Welcome to "+appName;	//not view name. it is response body
	}
}

SOAP vs REST
------------------
REST vs. SOAP
=============

1. Protocol:
   - REST: Uses HTTP only
   - SOAP: Uses HTTP, SMTP, TCP, etc.

2. Data Format:
   - REST: Supports JSON, XML, plain text, HTML
   - SOAP: Supports only XML

3. Type:
   - REST: Architectural style
   - SOAP: Protocol (strict standard)

4. Speed:
   - REST: Faster (lightweight)
   - SOAP: Slower (verbose XML)

5. Flexibility:
   - REST: Highly flexible
   - SOAP: Strict and rigid

6. Ease of Use:
   - REST: Easy to test (Postman, curl)
   - SOAP: Requires SOAP tools (e.g., SoapUI)

7. Security:
   - REST: Basic (HTTPS, OAuth)
   - SOAP: Advanced (WS-Security, encryption, signatures)

8. Standards:
   - REST: No strict standards
   - SOAP: WSDL, XSD, WS-Security, etc.

9. Stateless:
   - REST: Stateless only
   - SOAP: Can be stateless or stateful

10. Usage:
    - REST: Web/mobile apps, microservices
    - SOAP: Enterprise, legacy systems, banking, telecom

Summary:

- REST is lightweight, flexible, and modern (preferred for new applications)
- SOAP is formal, secure, and used in enterprise systems with complex requirements

import java.io.IOException;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.time.Duration;

public class APICall {
   public static void main(String[] args) {
      HttpClient httpClient = HttpClient.newBuilder()
         .version(HttpClient.Version.HTTP_2)
         .connectTimeout(Duration.ofSeconds(10))
         .build(); 
         try {
            HttpRequest request = HttpRequest.newBuilder()
            .GET()
            .uri(URI.create("http://localhost:8085/person"))
            .build();                              
            HttpResponse<String> response = httpClient.send(request,
            HttpResponse.BodyHandlers.ofString()); 

         System.out.println("Status code: " + response.statusCode());                            
         System.out.println("Headers: " + response.headers().allValues("content-type"));
         System.out.println("Body: " + response.body());
      } catch (IOException | InterruptedException e) {
         e.printStackTrace();
      }
   }
}

@RequestBody vs @ResponseBody

@RequestBody
	when we send data to a restful webservice via body, and not in URL

	if you are sending via URL, then it can be
		path variable (or)
		request param


the webservice is sending response, ie., Response body

A method in rest controller returns @ResponseBody by default. So we do not mention it.


A method in a MVC project returns view or model and view. If we want to modify that into response body, then we use @ResponseBody annotation.

Spring MVC REST Annotations
===========================

1. @RequestMapping
-------------------
- Maps HTTP requests to handler methods.

Example:
@RequestMapping(value = "/hello", method = RequestMethod.GET)
public String sayHello() {
    return "Hello World!";
}

Tip: @GetMapping, @PostMapping, @PutMapping, etc., are shortcuts for @RequestMapping.

---

2. @RequestBody
----------------
- Binds the request body (usually JSON) to a Java object.

Example:
@PostMapping("/user")
public String saveUser(@RequestBody User user) {
    return "User saved: " + user.getName();
}

Request:
POST /user
Content-Type: application/json
{
  "name": "Jag",
  "email": "jag@example.com"
}

---

3. @ResponseBody
-----------------
- Sends the return value directly as the HTTP response (usually JSON).
- Implicit when using @RestController.

Example:
@ResponseBody
@GetMapping("/ping")
public String ping() {
    return "pong";
}

OR

@RestController
public class MyController {
    @GetMapping("/ping")
    public String ping() {
        return "pong";
    }
}

---

4. @PathVariable
-----------------
- Binds a variable from the URL path.

Example:
@GetMapping("/greet/{name}")
public String greet(@PathVariable String name) {
    return "Hello, " + name;
}

Request: GET /greet/Jag
Response: Hello, Jag

---

5. @RequestParam
-----------------
- Binds query parameter value to a method parameter.

Example:
@GetMapping("/search")
public String search(@RequestParam String keyword) {
    return "Searching for: " + keyword;
}

Request: GET /search?keyword=spring
Response: Searching for: spring

---

6. @RequestHeader
------------------
- Binds an HTTP header value to a method parameter.

Example:
@GetMapping("/agent")
public String getAgent(@RequestHeader("User-Agent") String agent) {
    return "Agent: " + agent;
}

Request: GET /agent
Header: User-Agent: PostmanRuntime/7.36.1
Response: Agent: PostmanRuntime/7.36.1

---

Summary
-------
Annotation        Purpose
----------        ---------------------------------------
@RequestMapping    Maps URL + method to controller method
@RequestBody       Converts request JSON to Java object
@ResponseBody      Converts Java object to JSON response
@PathVariable      Extracts value from URL path
@RequestParam      Extracts value from query parameter
@RequestHeader     Extracts value from HTTP headers

package com.harman.demo.controller;

import java.util.List;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.DeleteMapping;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import com.harman.demo.entity.Product;
import com.harman.demo.service.ProductService;

@RestController
@RequestMapping("/product")
public class ProductController {
	@Autowired
	private ProductService ps;
	
	@PostMapping
	public Product create(@RequestBody Product product) {
		return ps.create(product);
	}
	
	@GetMapping
	public List<Product> read() {
		return ps.read();
	}
	
	@GetMapping("/{id}")
	public Product read(@PathVariable("id") Long id) {
		return ps.read(id);
	}
	
	@PutMapping("/{id}")
	public Product update(@PathVariable Long id, @RequestBody Product product) {
		return ps.update(id, product);
	}
	
	@DeleteMapping("/{id}")
	public Product delete(@PathVariable Long id) {
		return ps.delete(id);
	}
	
}
